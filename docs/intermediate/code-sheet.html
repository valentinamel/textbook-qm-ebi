analysis <- subset(dat, x > 0 & group == "A")

Lecture 1: Simple linear regression

Population model and fitted line

\[ Y_i=\beta_0+\beta_1X_i+\varepsilon_i, \qquad \widehat Y_i=\hat\beta_0+\hat\beta_1X_i. \]

OLS estimates in simple regression

\[ \hat\beta_1= \frac{\sum_i(x_i-\bar x)(y_i-\bar y)} {\sum_i(x_i-\bar x)^2} =\frac{s_{XY}}{s_X^2} =r_{XY}\,\frac{s_Y}{s_X}, \qquad \hat\beta_0=\bar y-\hat\beta_1\bar x. \]

Residual and sum of squared residuals

\[ e_i=Y_i-\widehat Y_i, \qquad \operatorname{SSE}=\sum_{i=1}^{n}e_i^2. \]

Change of the units of measurement by a factor \(c\)

Multiplied by \(c\) New intercept New slope
outcome only \(c\,\hat\beta_0\) \(c\,\hat\beta_1\)
predictor only \(\hat\beta_0\) \(\hat\beta_1/c\)
outcome and predictor \(c\,\hat\beta_0\) \(\hat\beta_1\)

Fit a simple regression

m <- lm(y ~ x, data = dat)
coef(m)          # estimates
summary(m)       # coefficient table and fit
fitted(m)        # fitted values
residuals(m)     # residuals

Scatter plot with the fitted line

plot(y ~ x, data = dat,
     xlab = "Predictor (unit)", ylab = "Outcome (unit)")
abline(m, col = "blue", lwd = 2)

Fitted values for chosen values x_A and x_B of the predictor

b <- coef(m)
b[1] + b[2] * x_A                  # fitted value by hand

new_cases <- data.frame(x = c(x_A, x_B))
predict(m, newdata = new_cases)    # fitted values

Lecture 2: What the model assumes and how residuals reveal it

Residual plot and normal Q–Q plot

plot(fitted(m), residuals(m),
     xlab = "Fitted value", ylab = "Residual")
abline(h = 0, lty = 2)

qqnorm(rstandard(m))
qqline(rstandard(m))

Standardised residuals and Cook’s distance

summary(residuals(m))    # summary of the residuals
rstandard(m)             # standardised residuals
cooks.distance(m)        # Cook's distances
nobs(m)                  # number of observations n

Flagged observations

\[ |\text{standardised residual}|>3, \qquad \text{Cook's distance}>\frac{4}{n}. \]

sum(abs(rstandard(m)) > 3)            # how many beyond ±3
sum(cooks.distance(m) > 4 / nobs(m))  # how many above 4/n

Lecture 3: Coefficient inference, fit, and the F-test

Two-sided test of one coefficient

\[ H_0:\beta_j=0, \qquad H_1:\beta_j\ne0, \qquad t=\frac{\hat\beta_j}{\operatorname{SE}(\hat\beta_j)}. \]

\[ \text{Reject }H_0\text{ at the 5\% level if }|t_{obs}|\ge t_{0.975,\,df}. \]

Degrees of freedom and 95% confidence interval, model with \(k\) slopes and an intercept

\[ df=n-k-1, \qquad \hat\beta_j\pm t_{0.975,\,df}\,\operatorname{SE}(\hat\beta_j), \qquad \text{simple regression: }k=1,\ df=n-2. \]

summary(m)                  # coefficient table and fit
confint(m, level = 0.95)    # 95% confidence intervals
df.residual(m)              # residual degrees of freedom
qt(0.975, df.residual(m))   # critical value, 95% interval

Labels in the output of summary(m)

Label in the output Symbol
Estimate \(\hat\beta_j\)
Std. Error \(\operatorname{SE}(\hat\beta_j)\)
t value \(t\)
Pr(>|t|) two-sided p-value for \(H_0:\beta_j=0\)
Residual standard error \(s\), with \(df=n-k-1\)
Multiple R-squared \(R^2\)
Adjusted R-squared \(R^2_{\text{adjusted}}\)
F-statistic overall \(F\), with \(k\) and \(n-k-1\) degrees of freedom

The t statistic and the two-sided p-value step by step

tab <- summary(m)$coefficients
t_value <- tab["x", "Estimate"] / tab["x", "Std. Error"]
p_value <- 2 * pt(-abs(t_value), df = df.residual(m))

Sums of squares and \(R^2\)

\[ \text{SST}=\sum_i(y_i-\bar y)^2,\qquad \text{SSE}=\sum_i(y_i-\hat y_i)^2,\qquad \text{SSR}=\sum_i(\hat y_i-\bar y)^2, \]

\[ \text{SST}=\text{SSR}+\text{SSE}, \qquad R^2=1-\frac{\text{SSE}}{\text{SST}} =\frac{\text{SSR}}{\text{SST}}. \]

y <- dat$y            # outcome values
yhat <- fitted(m)    # fitted values

SST <- sum((y - mean(y))^2)
SSE <- sum(residuals(m)^2)
SSR <- SST - SSE
R2 <- 1 - SSE / SST

Residual standard error and adjusted \(R^2\)

\[ s=\sqrt{\frac{\text{SSE}}{n-k-1}}, \qquad R^2_{\text{adjusted}} =1-(1-R^2)\frac{n-1}{n-k-1}. \]

summary(m)$r.squared        # R squared
summary(m)$fstatistic       # overall F, k, and n - k - 1

Overall F-test, model with \(k\) slopes

\[ H_0:\beta_1=\cdots=\beta_k=0, \qquad H_1:\text{at least one of }\beta_1,\ldots,\beta_k\text{ is not }0, \]

\[ F=\frac{\text{SSR}/k}{\text{SSE}/(n-k-1)}, \qquad \text{simple regression: }F=t^2. \]

Heteroskedasticity-robust standard errors (HC1)

library(lmtest)
library(sandwich)

# robust standard errors, t statistics, and p-values
coeftest(m, vcov. = vcovHC(m, type = "HC1"))
# robust 95% confidence intervals
coefci(m, vcov. = vcovHC(m, type = "HC1"))

Lecture 4: Multiple regression and what controls do

Model with \(k\) predictors

\[ Y_i=\beta_0+\beta_1X_{i1}+\cdots+\beta_kX_{ik}+\varepsilon_i. \]

m_multi <- lm(y ~ x1 + x2 + x3, data = analysis)
summary(m_multi)     # coefficient table and fit
confint(m_multi)     # 95% confidence intervals

Fitted difference for changes \(\Delta x_1,\ldots,\Delta x_k\) of the predictors

\[ \Delta\widehat Y=\hat\beta_1\Delta x_1+\cdots+\hat\beta_k\Delta x_k. \]

Fitted difference between two cases A and B

cases <- data.frame(
  x1 = c(x1_A, x1_B),
  x2 = c(x2_A, x2_B),
  x3 = c(x3_A, x3_B)
)
fits <- predict(m_multi, newdata = cases)  # fitted values
fits[2] - fits[1]                          # B minus A

Regression table with one column per model

# install once: install.packages("modelsummary")
library(modelsummary)
m_a <- lm(y ~ x1, data = analysis)
m_b <- lm(y ~ x1 + x2, data = analysis)
modelsummary(
  list("(1)" = m_a, "(2)" = m_b, "(3)" = m_multi),
  stars = c("*" = 0.05, "**" = 0.01, "***" = 0.001),
  gof_map = c("nobs", "r.squared", "adj.r.squared")
)

Lecture 5: Categories as predictors and joint tests

Reference category and model with a categorical predictor

analysis$group <- factor(analysis$group)
# categories: the first one is the reference
levels(analysis$group)
analysis$group <- relevel(analysis$group,
                          ref = "reference")

m_group <- lm(y ~ x1 + x2 + group, data = analysis)
summary(m_group)
coef(m_group)
confint(m_group)

The same model with another reference category

analysis$group_alt <- relevel(analysis$group,
                              ref = "other")
m_alt <- lm(y ~ x1 + x2 + group_alt, data = analysis)
# TRUE if the fitted values of the two models are equal
all.equal(fitted(m_group), fitted(m_alt))

Partial F-test of nested models

reduced <- lm(y ~ x1 + x2, data = analysis)
complete <- lm(y ~ x1 + x2 + group, data = analysis)
anova(reduced, complete)

Complete (unrestricted) model: \(k_C\) slopes. Reduced (restricted) model: the complete model without \(q\) of its terms, with the same outcome and the same observations.

\[ H_0:\text{the }q\text{ coefficients left out of the reduced model are all }0, \]

\[ F= \frac{(\text{SSE}_{R}-\text{SSE}_{C})/q} {\text{SSE}_{C}/(n-k_C-1)}. \]

Labels in the output of anova(reduced, complete)

Label in the output Symbol
Res.Df \(n-k-1\) of each model
RSS \(\text{SSE}_R\) in row 1, \(\text{SSE}_C\) in row 2
Df \(q\)
Sum of Sq \(\text{SSE}_R-\text{SSE}_C\)
F partial \(F\)
Pr(>F) p-value of the partial F-test

One-way ANOVA: one model, \(G\) groups

m_anova <- lm(y ~ group, data = analysis)
anova(m_anova)   # ANOVA table

\[ H_0:\text{all }G\text{ group means are equal}, \qquad F=\frac{\text{SSR}/(G-1)}{\text{SSE}/(n-G)}. \]

Labels in the output of anova(m_anova)

Label in the output Symbol
Df \(G-1\) (group), \(n-G\) (Residuals)
Sum Sq SSR (group), SSE (Residuals)
Mean Sq Sum Sq / Df
F value \(F\)
Pr(>F) p-value of the F-test

Lecture 6: Interactions and curvature

Interaction of a numerical predictor \(X\) with a dummy \(D\)

\[ \widehat Y=\hat\beta_0+\hat\beta_1X+\hat\beta_2D+\hat\beta_3XD. \]

Group Intercept Slope
\(D=0\) \(\hat\beta_0\) \(\hat\beta_1\)
\(D=1\) \(\hat\beta_0+\hat\beta_2\) \(\hat\beta_1+\hat\beta_3\) 
m_interaction <- lm(y ~ x * group + z, data = analysis)
# x * group expands to x + group + x:group
summary(m_interaction)

Centred predictor

# x minus its mean
analysis$x_c <- analysis$x - mean(analysis$x)
m_centred <- lm(y ~ x_c * group + z, data = analysis)

Quadratic model

\[ \widehat Y=\hat\beta_0+\hat\beta_1X+\hat\beta_2X^2. \]

\[ \text{Fitted change from }x\text{ to }x+1:\quad \hat\beta_1+\hat\beta_2(2x+1). \]

m_linear <- lm(y ~ x + z, data = analysis)
m_quadratic <- lm(y ~ x + I(x^2) + z, data = analysis)
# I(x^2) is the square of x

summary(m_quadratic)
anova(m_linear, m_quadratic)
